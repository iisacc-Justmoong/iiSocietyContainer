cmake_minimum_required(VERSION 3.24)
project(SocietySharingConsumer LANGUAGES CXX)
set(CMAKE_DISABLE_FIND_PACKAGE_Qt6Qml TRUE)
set(CMAKE_DISABLE_FIND_PACKAGE_Qt6Quick TRUE)
find_package(iiSocietyContainer CONFIG REQUIRED COMPONENTS Sharing)
if(TARGET Qt6::Qml OR TARGET Qt6::Quick)
    message(FATAL_ERROR "Sharing imported a Quick presentation dependency")
endif()
include(CTest)
add_executable(sharing_consumer main.cpp)
target_link_libraries(sharing_consumer PRIVATE iiSocietyContainer::Sharing)
add_test(NAME NativeStorage.InstalledSharing COMMAND sharing_consumer)
