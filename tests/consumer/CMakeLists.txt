cmake_minimum_required(VERSION 3.24)
project(iiSocietyContainerConsumer LANGUAGES CXX)

option(IISOCIETYCONTAINER_TEST_GUI "Verify the installed GUI component headers" OFF)
if(IISOCIETYCONTAINER_TEST_GUI)
    find_package(iiSocietyContainer 0.15.0 CONFIG REQUIRED COMPONENTS Gui)
else()
    find_package(iiSocietyContainer 0.15.0 CONFIG REQUIRED)
endif()
if(APPLE AND NOT CMAKE_SYSTEM_NAME STREQUAL "iOS")
    get_filename_component(society_native_bundle_name "${iiSocietyContainer_NATIVE_APP}" NAME)
    if(NOT society_native_bundle_name STREQUAL "Society.app")
        message(FATAL_ERROR "The installed Finder adapter must be named Society.app")
    endif()
endif()
if(NOT COMMAND iiSocietyContainer_add_ios_file_provider OR NOT COMMAND iiSocietyContainer_configure_ios_client)
    message(FATAL_ERROR "The installed package must include its iOS integration function")
endif()
if(NOT COMMAND iiSocietyContainer_add_android_file_provider OR NOT COMMAND iiSocietyContainer_configure_android_client)
    message(FATAL_ERROR "The installed package must include Android owner and client integration")
endif()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(_society_consumer_runner "${Python3_EXECUTABLE}" -B
    "${CMAKE_CURRENT_LIST_DIR}/../resource_runner.py" auto "${CMAKE_CURRENT_LIST_DIR}/../../resources")
add_executable(iiSocietyContainerConsumer ../hello_world.cpp)
target_link_libraries(iiSocietyContainerConsumer PRIVATE iiSocietyContainer::iiSocietyContainer)
set_target_properties(iiSocietyContainerConsumer PROPERTIES CXX_EXTENSIONS OFF)
enable_testing()
add_executable(iiSocietyContainerManagedConsumer ../managed_storage.cpp)
target_link_libraries(iiSocietyContainerManagedConsumer PRIVATE iiSocietyContainer::iiSocietyContainer)
add_test(NAME iiSocietyContainer.installed_managed_storage COMMAND iiSocietyContainerManagedConsumer)
add_test(NAME iiSocietyContainer.installed_consumer COMMAND iiSocietyContainerConsumer)

add_executable(iiSocietyContainerDirectoryConsumer ../society_container.cpp)
target_link_libraries(iiSocietyContainerDirectoryConsumer PRIVATE
    iiSocietyContainer::iiSocietyContainer)
set_target_properties(iiSocietyContainerDirectoryConsumer PROPERTIES AUTOMOC ON CXX_EXTENSIONS OFF)
add_test(NAME iiSocietyContainer.installed_directory_space COMMAND ${_society_consumer_runner} $<TARGET_FILE:iiSocietyContainerDirectoryConsumer>)

add_executable(iiSocietyContainerDriveConsumer ../society_drive.cpp)
target_link_libraries(iiSocietyContainerDriveConsumer PRIVATE iiSocietyContainer::iiSocietyContainer)
set_target_properties(iiSocietyContainerDriveConsumer PROPERTIES AUTOMOC ON CXX_EXTENSIONS OFF)
add_test(NAME iiSocietyContainer.installed_drive COMMAND ${_society_consumer_runner} $<TARGET_FILE:iiSocietyContainerDriveConsumer>)
add_test(NAME iiSocietyContainer.installed_drive_safe COMMAND ${_society_consumer_runner} $<TARGET_FILE:iiSocietyContainerDriveConsumer> rejectsInvalidManifestWithoutOverwritingIt)

add_executable(iiSocietyContainerStorageConsumer ../shared_storage.cpp)
target_link_libraries(iiSocietyContainerStorageConsumer PRIVATE iiSocietyContainer::iiSocietyContainer iiFileProvider::Native)
set_target_properties(iiSocietyContainerStorageConsumer PROPERTIES AUTOMOC ON CXX_EXTENSIONS OFF)
add_test(NAME iiSocietyContainer.installed_shared_storage COMMAND ${_society_consumer_runner} $<TARGET_FILE:iiSocietyContainerStorageConsumer>)
set_tests_properties(iiSocietyContainer.installed_shared_storage PROPERTIES SKIP_RETURN_CODE 77)
add_test(NAME iiSocietyContainer.installed_shared_storage_model_contract COMMAND ${_society_consumer_runner} $<TARGET_FILE:iiSocietyContainerStorageConsumer> directoryRefreshPublishesOnlyTheChangedRows)
set_tests_properties(iiSocietyContainer.installed_shared_storage_model_contract PROPERTIES TIMEOUT 120)
add_executable(iiSocietyContainerLogicalConsumer ../logical_storage.cpp)
target_link_libraries(iiSocietyContainerLogicalConsumer PRIVATE iiSocietyContainer::iiSocietyContainer)
set_target_properties(iiSocietyContainerLogicalConsumer PROPERTIES AUTOMOC ON CXX_EXTENSIONS OFF)
add_test(NAME iiSocietyContainer.installed_logical_storage COMMAND ${_society_consumer_runner} $<TARGET_FILE:iiSocietyContainerLogicalConsumer>)
set_tests_properties(iiSocietyContainer.installed_logical_storage PROPERTIES TIMEOUT 300 SKIP_RETURN_CODE 77)
add_executable(iiSocietyContainerFilesConsumer ../files_view.cpp)
target_link_libraries(iiSocietyContainerFilesConsumer PRIVATE iiSocietyContainer::iiSocietyContainer)
set_target_properties(iiSocietyContainerFilesConsumer PROPERTIES AUTOMOC ON CXX_EXTENSIONS OFF)
add_test(NAME iiSocietyContainer.installed_files_projection COMMAND ${_society_consumer_runner} $<TARGET_FILE:iiSocietyContainerFilesConsumer>)

add_executable(iiSocietyContainerModelsConsumer ../model_store.cpp)
target_link_libraries(iiSocietyContainerModelsConsumer PRIVATE iiSocietyContainer::iiSocietyContainer iiFileProvider::Native)
set_target_properties(iiSocietyContainerModelsConsumer PROPERTIES AUTOMOC ON CXX_EXTENSIONS OFF)
add_test(NAME iiSocietyContainer.installed_models COMMAND ${_society_consumer_runner} $<TARGET_FILE:iiSocietyContainerModelsConsumer>)
add_test(NAME iiSocietyContainer.installed_models_invalid_tensor COMMAND ${_society_consumer_runner} $<TARGET_FILE:iiSocietyContainerModelsConsumer> validatesTensorDescriptorsAndPayload:escaped-duplicate-field)

add_executable(iiSocietyContainerFileTreeConsumer ../file_tree.cpp)
target_link_libraries(iiSocietyContainerFileTreeConsumer PRIVATE iiSocietyContainer::FileTree)
add_test(NAME iiSocietyContainer.installed_file_tree COMMAND iiSocietyContainerFileTreeConsumer "${CMAKE_CURRENT_BINARY_DIR}")

add_executable(iiSocietyContainerOperationsConsumer ../file_operations.cpp
    ../../src/ExclusiveRename.cpp ../../src/AssetTransfer.cpp ../../src/DeletedOrigins.cpp)
target_include_directories(iiSocietyContainerOperationsConsumer PRIVATE "${CMAKE_CURRENT_LIST_DIR}/../..")
target_link_libraries(iiSocietyContainerOperationsConsumer PRIVATE iiSocietyContainer::iiSocietyContainer)
set_target_properties(iiSocietyContainerOperationsConsumer PROPERTIES AUTOMOC ON CXX_EXTENSIONS OFF)
add_test(NAME iiSocietyContainer.installed_file_operations COMMAND ${_society_consumer_runner} $<TARGET_FILE:iiSocietyContainerOperationsConsumer>)

set_tests_properties(iiSocietyContainer.installed_directory_space iiSocietyContainer.installed_files_projection iiSocietyContainer.installed_drive iiSocietyContainer.installed_drive_safe
    iiSocietyContainer.installed_file_operations PROPERTIES SKIP_RETURN_CODE 77)

if(IISOCIETYCONTAINER_TEST_GUI)
    add_executable(iiSocietyContainerGuiHeaders gui_headers.cpp)
    target_link_libraries(iiSocietyContainerGuiHeaders PRIVATE iiSocietyContainer::Gui)
    add_test(NAME iiSocietyContainer.installed_gui_headers COMMAND iiSocietyContainerGuiHeaders)
endif()
