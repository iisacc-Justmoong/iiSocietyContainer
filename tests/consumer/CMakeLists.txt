cmake_minimum_required(VERSION 3.24)
project(iiSocietyContainerConsumer LANGUAGES CXX)

find_package(iiSocietyContainer 0.12.0 CONFIG REQUIRED)
if(APPLE AND NOT CMAKE_SYSTEM_NAME STREQUAL "iOS")
    get_filename_component(society_native_bundle_name "${iiSocietyContainer_NATIVE_APP}" NAME)
    if(NOT society_native_bundle_name STREQUAL "Society.app")
        message(FATAL_ERROR "The installed Finder adapter must be named Society.app")
    endif()
endif()
if(NOT COMMAND iiSocietyContainer_add_ios_file_provider OR NOT COMMAND iiSocietyContainer_configure_ios_client)
    message(FATAL_ERROR "The installed package must include its iOS integration function")
endif()
if(NOT COMMAND iiSocietyContainer_add_android_file_provider OR NOT COMMAND iiSocietyContainer_configure_android_client)
    message(FATAL_ERROR "The installed package must include Android owner and client integration")
endif()
find_package(Qt6 6.8.3 EXACT REQUIRED COMPONENTS Test)
add_executable(iiSocietyContainerConsumer ../hello_world.cpp)
target_link_libraries(iiSocietyContainerConsumer PRIVATE iiSocietyContainer::iiSocietyContainer)
set_target_properties(iiSocietyContainerConsumer PROPERTIES CXX_EXTENSIONS OFF)
enable_testing()
add_test(NAME iiSocietyContainer.installed_consumer COMMAND iiSocietyContainerConsumer)

add_executable(iiSocietyContainerDirectoryConsumer ../society_container.cpp)
target_link_libraries(iiSocietyContainerDirectoryConsumer PRIVATE
    iiSocietyContainer::iiSocietyContainer Qt6::Test)
set_target_properties(iiSocietyContainerDirectoryConsumer PROPERTIES AUTOMOC ON CXX_EXTENSIONS OFF)
add_test(NAME iiSocietyContainer.installed_directory_space COMMAND iiSocietyContainerDirectoryConsumer)

add_executable(iiSocietyContainerDriveConsumer ../society_drive.cpp)
target_link_libraries(iiSocietyContainerDriveConsumer PRIVATE iiSocietyContainer::iiSocietyContainer Qt6::Test)
set_target_properties(iiSocietyContainerDriveConsumer PROPERTIES AUTOMOC ON CXX_EXTENSIONS OFF)
add_test(NAME iiSocietyContainer.installed_drive COMMAND iiSocietyContainerDriveConsumer)

add_executable(iiSocietyContainerStorageConsumer ../shared_storage.cpp)
target_link_libraries(iiSocietyContainerStorageConsumer PRIVATE iiSocietyContainer::iiSocietyContainer Qt6::Test)
set_target_properties(iiSocietyContainerStorageConsumer PROPERTIES AUTOMOC ON CXX_EXTENSIONS OFF)
add_test(NAME iiSocietyContainer.installed_shared_storage COMMAND iiSocietyContainerStorageConsumer)
add_executable(iiSocietyContainerFilesConsumer ../files_view.cpp)
target_link_libraries(iiSocietyContainerFilesConsumer PRIVATE iiSocietyContainer::iiSocietyContainer Qt6::Test)
target_compile_definitions(iiSocietyContainerFilesConsumer PRIVATE SOCIETY_TEST_DIRECTORY="${CMAKE_CURRENT_BINARY_DIR}")
set_target_properties(iiSocietyContainerFilesConsumer PROPERTIES AUTOMOC ON CXX_EXTENSIONS OFF)
add_test(NAME iiSocietyContainer.installed_files_projection COMMAND iiSocietyContainerFilesConsumer)

add_executable(iiSocietyContainerModelsConsumer ../model_store.cpp)
target_link_libraries(iiSocietyContainerModelsConsumer PRIVATE iiSocietyContainer::iiSocietyContainer Qt6::Test)
target_compile_definitions(iiSocietyContainerModelsConsumer PRIVATE SOCIETY_TEST_DIRECTORY="${CMAKE_CURRENT_BINARY_DIR}")
set_target_properties(iiSocietyContainerModelsConsumer PROPERTIES AUTOMOC ON CXX_EXTENSIONS OFF)
add_test(NAME iiSocietyContainer.installed_models COMMAND iiSocietyContainerModelsConsumer)
