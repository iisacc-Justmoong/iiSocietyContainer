cmake_minimum_required(VERSION 3.24)
project(iiSocietyContainerNativeTreeConsumer LANGUAGES CXX)
find_package(iiSocietyContainer 0.15.0 CONFIG REQUIRED COMPONENTS FileTree)
add_executable(NativeTreeConsumer ../../file_tree.cpp)
target_link_libraries(NativeTreeConsumer PRIVATE iiSocietyContainer::FileTree)
enable_testing()
add_test(NAME iiSocietyContainer.qt_free_file_tree COMMAND NativeTreeConsumer "${CMAKE_CURRENT_BINARY_DIR}")
add_executable(NativeDirectoryConsumer ../../directory_storage.cpp)
target_link_libraries(NativeDirectoryConsumer PRIVATE iiSocietyContainer::FileTree)
add_test(NAME iiSocietyContainer.qt_free_directory_storage COMMAND NativeDirectoryConsumer "${CMAKE_CURRENT_BINARY_DIR}")
